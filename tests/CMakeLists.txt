cmake_minimum_required(VERSION 3.22)
project(infantry_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

function(add_logic_test name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ../Modules ../Application)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()

add_logic_test(pid_test pid_test.cpp ../Modules/algorithm/pid.cpp)

target_include_directories(gimbal_calibration_test BEFORE PRIVATE stubs ../BSP)
